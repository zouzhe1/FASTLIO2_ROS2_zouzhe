cmake_minimum_required(VERSION 3.8)
project(place_recognition)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(place_index src/place_index.cpp)
target_include_directories(place_index PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(place_index yaml-cpp)
install(TARGETS place_index EXPORT place_index_targets)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(place_index_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(yaml-cpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_place_index test/test_place_index.cpp)
  target_link_libraries(test_place_index place_index)
endif()
ament_package()
